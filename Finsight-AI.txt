function UploadZone({ onUpload, loading }) {
  const [drag, setDrag] = useState(false);
  const [file, setFile] = useState(null);
  const [text, setText] = useState("");
  const fileRef = useRef();

  const readFile = (f) => {
    setFile(f);
    const reader = new FileReader();
    reader.onload = (e) => {
      const arr = new Uint8Array(e.target.result);
      let str = "";
      for (let i = 0; i < Math.min(arr.length, 60000); i++) {
        if (arr[i] > 31 && arr[i] < 127) str += String.fromCharCode(arr[i]);
        else if (arr[i] === 10 || arr[i] === 13) str += "\n";
      }
      str = str.replace(/[^\x20-\x7E\n]/g, " ").replace(/ {3,}/g, " ").trim();
      setText(str.slice(0, 12000));
    };
    reader.readAsArrayBuffer(f);
  };

  const onDrop = useCallback((e) => {
    e.preventDefault(); setDrag(false);
    const f = e.dataTransfer.files[0];
    if (f && f.type === "application/pdf") readFile(f);
  }, []);

  return (
    <div style={{ maxWidth: 680, margin: "0 auto" }}>
      <div
        onDrop={onDrop}
        onDragOver={(e) => { e.preventDefault(); setDrag(true); }}
        onDragLeave={() => setDrag(false)}
        onClick={() => fileRef.current.click()}
        style={{
          border: `2px dashed ${drag ? COLORS.emerald : COLORS.gray}`,
          borderRadius: 16, padding: "48px 32px", textAlign: "center", cursor: "pointer",
          background: drag ? "rgba(0,168,107,0.04)" : "#fff", transition: "all 0.2s"
        }}
      >
        <div style={{ fontSize: 40, marginBottom: 16 }}>📂</div>
        <h3 style={{ fontWeight: 700, color: COLORS.navy, marginBottom: 8 }}>
          {file ? file.name : "Drop your PDF here"}
        </h3>
        <p style={{ color: COLORS.muted, fontSize: 14 }}>
          {file ? `${(file.size / 1024).toFixed(0)} KB ready` : "Annual reports, earnings transcripts, 10-K, 10-Q files"}
        </p>
        <input ref={fileRef} type="file" accept=".pdf" style={{ display: "none" }} onChange={(e) => { if (e.target.files[0]) readFile(e.target.files[0]); }} />
      </div>

      <div style={{ marginTop: 16 }}>
        <label style={{ fontSize: 13, color: COLORS.muted, fontWeight: 600 }}>Or paste report text / description:</label>
        <textarea
          value={text}
          onChange={(e) => setText(e.target.value)}
          placeholder="Paste annual report text, financial highlights, or describe a company for demo analysis…"
          style={{ width: "100%", minHeight: 120, marginTop: 8, borderRadius: 10, border: `1px solid ${COLORS.gray}`, padding: "12px 14px", fontSize: 14, color: COLORS.text, resize: "vertical", boxSizing: "border-box" }}
        />
      </div>

      <button
        onClick={() => onUpload(text || `Demo analysis for a large-cap technology company with strong revenue growth, cloud segment expansion, and some regulatory risks. Analyze as if this is a real annual report with the following context: Strong revenue growth of 15% YoY, expanding margins, cloud segment leading growth, some regulatory headwinds in EU, robust cash generation, dividend growth.`)}
        disabled={loading}
        style={{
          width: "100%", marginTop: 16, background: loading ? "#94a3b8" : COLORS.emerald,
          color: "#fff", border: "none", borderRadius: 10, padding: "14px", fontSize: 16,
          fontWeight: 700, cursor: loading ? "not-allowed" : "pointer"
        }}
      >
        {loading ? "🔄 Analyzing with AI…" : "⚡ Generate Financial Analysis"}
      </button>
    </div>
  );
}

function SentimentGauge({ score }) {
  const pct = Math.max(0, Math.min(100, score));
  const color = pct > 65 ? COLORS.emerald : pct > 40 ? COLORS.gold : "#ef4444";
  const label = pct > 65 ? "Positive" : pct > 40 ? "Neutral" : "Negative";
  return (
    <div style={{ textAlign: "center" }}>
      <div style={{ position: "relative", display: "inline-block", width: 160, height: 80 }}>
        <svg width="160" height="90" viewBox="0 0 160 90">
          <path d="M 20 80 A 60 60 0 0 1 140 80" fill="none" stroke={COLORS.gray} strokeWidth="12" strokeLinecap="round" />
          <path d="M 20 80 A 60 60 0 0 1 140 80" fill="none" stroke={color} strokeWidth="12" strokeLinecap="round"
            strokeDasharray={`${(pct / 100) * 188} 188`} />
          <text x="80" y="72" textAnchor="middle" fontSize="22" fontWeight="800" fill={color}>{pct}</text>
          <text x="80" y="86" textAnchor="middle" fontSize="12" fill={COLORS.muted}>{label}</text>

  );
}

export default function App() {
  const [view, setView] = useState("landing");
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState(null);
  const [error, setError] = useState("");

  const analyze = async (text) => {
    setLoading(true); setError("");
    try {
      const res = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          model: "claude-sonnet-4-20250514",
          max_tokens: 1000,
          messages: [{ role: "user", content: PROMPT(text) }]
        })
      });
      const data = await res.json();
      const raw = data.content?.[0]?.text || "";
      const clean = raw.replace(/```json|```/g, "").trim();
      const parsed = JSON.parse(clean);
      setResult(parsed); setView("dashboard");
    } catch (e) {
      setError("Analysis failed. Please try again or paste text directly.");
    }
    setLoading(false);
  };

  return (
    <div style={S.app}>
      <Navbar onAnalyze={() => setView("upload")} />

      {view === "landing" && (
        <>
          <Hero onAnalyze={() => setView("upload")} />
          <Features />
          <HowItWorks />
          <section style={{ padding: "60px 2rem", textAlign: "center", background: "#fff" }}>
            <h2 style={{ fontSize: 30, fontWeight: 800, color: COLORS.navy, marginBottom: 12 }}>Ready to analyze your first report?</h2>
            <p style={{ color: COLORS.muted, marginBottom: 28 }}>Upload any PDF or paste text — results in under 30 seconds.</p>
            <button onClick={() => setView("upload")} style={{ background: COLORS.emerald, color: "#fff", border: "none", borderRadius: 10, padding: "14px 36px", fontSize: 16, fontWeight: 700, cursor: "pointer" }}>
              Start Free Analysis →
            </button>
          </section>
        </>
      )}

      {view === "upload" && (
        <section style={{ padding: "60px 2rem", maxWidth: 760, margin: "0 auto" }}>
          <div style={{ textAlign: "center", marginBottom: 36 }}>
            <h2 style={{ fontSize: 28, fontWeight: 800, color: COLORS.navy, marginBottom: 8 }}>Upload Financial Report</h2>
            <p style={{ color: COLORS.muted }}>Supports annual reports, 10-K, 10-Q, earnings transcripts</p>
          </div>
          {error && <div style={{ background: "#fee2e2", color: "#7f1d1d", borderRadius: 8, padding: "12px 16px", marginBottom: 16, fontSize: 14 }}>{error}</div>}
          <UploadZone onUpload={analyze} loading={loading} />
          {loading && (
            <div style={{ textAlign: "center", marginTop: 32 }}>
              <div style={{ display: "inline-flex", flexDirection: "column", gap: 8, alignItems: "center" }}>
                {["Extracting document text…", "Running financial analysis…", "Generating investment insights…"].map((s, i) => (
                  <div key={i} style={{ fontSize: 14, color: COLORS.muted, opacity: 0.7 + i * 0.1 }}>{s}</div>
                ))}
              </div>
            </div>
          )}
        </section>
      )}

      {view === "dashboard" && result && (
        <section style={{ padding: "40px 2rem" }}>
          <Dashboard data={result} onReset={() => { setView("upload"); setResult(null); }} />
        </section>
      )}

      <footer style={{ background: COLORS.navy, padding: "32px 2rem", marginTop: 40 }}>
        <div style={{ maxWidth: 1100, margin: "0 auto", display: "flex", justifyContent: "space-between", alignItems: "center", flexWrap: "wrap", gap: 16 }}>
          <div style={{ color: "#fff", fontWeight: 700, fontSize: 18 }}>FinSight <span style={{ color: COLORS.gold }}>AI</span></div>
          <div style={{ color: "rgba(255,255,255,0.4)", fontSize: 13 }}>© 2024 FinSight AI · For educational and analytical purposes only · Not financial advice</div>
        </div>
      </footer>
    </div>
  );
}
